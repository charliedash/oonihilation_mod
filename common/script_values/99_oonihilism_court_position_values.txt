# Salary values for tiger_wrangler_court_position (common/court_positions/types/99_oonihilism_court_positions.txt).
# Faithful copy of vanilla's own keeper_of_swans_total_salary_value / _prestige_salary_value
# (common/script_values/00_court_position_values.txt) with POS swapped to tiger_wrangler — this
# wrapper (inner circle perk, obligation hooks, halved/increased salary triggers, LAAMP freebies)
# is boilerplate every minor court position needs, not tiger-specific.
tiger_wrangler_total_salary_value = {
	value = minor_court_position_salary # Update X_salary_tooltip if script_value is changed
	# Set to 0 with Dynasty Perk Inner Circle
	if = {
		limit = { court_position_inner_circle_salary_trigger = { POS = tiger_wrangler } }
		multiply = {
			value = 0
			desc = inner_circle_salary_mod
			format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
		}
	}
	# Halved with obligation hook
	else_if = {
		limit = {
			court_position_obligation_hook_salary_trigger = {
				POS = tiger_wrangler
				LIEGE = scope:liege
			}
		}
		multiply = {
			value = 0.5
			desc = obligation_hook_salary
			format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
		}
	}
	# Halved salary
	if = {
		limit = {
			court_position_halved_salary_trigger = { POS = tiger_wrangler }
		}
		multiply = {
			value = 0.5
			desc = court_position_halved_salary
			format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
		}
	}
	# Increased salary
	if = {
		limit = {
			court_position_increased_salary_trigger = { POS = tiger_wrangler }
		}
		multiply = {
			value = 1.5
			desc = court_position_increased_salary
			format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
		}
	}
	#You're working with LAAMPS now
	if = {
		limit = {
			scope:liege ?= { government_has_flag = government_is_landless_adventurer }
		}
		multiply = {
			value = 0
			desc = court_position_laamps_free
			format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
		}
	}
}
