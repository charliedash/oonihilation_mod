# Supporting effects for the Pet Tiger story cycle — see common/story_cycles/99_oonihilism_pet_tiger.txt.
# Structurally modeled on vanilla's assign_eagle_gender_effect / remove_eagle_story_modifiers_effect /
# assign_name_options_eagle_story_cycle_effect (common/scripted_effects/00_animal_effects.txt), simplified:
# personality is a single permanent pick rather than a scaled/leveled trait. Naming now uses the same
# free-text entry widget as vanilla (common/events/oonihilism_events/99_oonihilism_pet_tiger_events.txt).

start_pet_tiger_story_cycle_effect = {
	show_as_tooltip = {
		add_character_modifier = {
			modifier = tiger_story_modifier
		}
	}
	hidden_effect = {
		create_story = story_cycle_pet_tiger
		# Permanent record of tiger ownership, unlike tiger_story_modifier/any_owned_story which both
		# clear once the story ends (tiger dies or is let go) — used by tiger_wrangler_court_position's
		# aptitude so the +30 bonus reflects having EVER had a pet tiger, not just currently owning one.
		add_character_flag = had_pet_tiger_story
	}
}

assign_tiger_gender_effect = {
	save_temporary_scope_value_as = {
		name = gender
		value = flag:$GENDER$
	}
	if = {
		limit = { scope:gender = flag:random }
		hidden_effect = {
			random_list = {
				50 = {
					set_variable = {
						name = tiger_gender
						value = flag:female
					}
				}
				50 = {
					set_variable = {
						name = tiger_gender
						value = flag:male
					}
				}
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				scope:gender = flag:male
				scope:gender = flag:female
			}
		}
		set_variable = {
			name = tiger_gender
			value = flag:$GENDER$
		}
	}
}

# Remove everything related to the story.
remove_tiger_story_modifiers_effect = {
	if = { limit = { has_character_modifier = tiger_story_modifier } remove_character_modifier = tiger_story_modifier }
	if = { limit = { has_character_modifier = fierce_tiger_story_modifier } remove_character_modifier = fierce_tiger_story_modifier }
	if = { limit = { has_character_modifier = loyal_tiger_story_modifier } remove_character_modifier = loyal_tiger_story_modifier }
	if = { limit = { has_character_modifier = calm_tiger_story_modifier } remove_character_modifier = calm_tiger_story_modifier }
	if = { limit = { has_character_modifier = cunning_tiger_story_modifier } remove_character_modifier = cunning_tiger_story_modifier }
	if = { limit = { has_character_modifier = well_fed_tiger_modifier } remove_character_modifier = well_fed_tiger_modifier }
	if = { limit = { has_character_modifier = feral_tiger_modifier } remove_character_modifier = feral_tiger_modifier }
	remove_character_flag = tiger_is_fierce
	remove_character_flag = tiger_is_loyal
	remove_character_flag = tiger_is_calm
	remove_character_flag = tiger_is_cunning
	remove_variable = story_cycle_tiger_name
	remove_variable = tiger_age_variable
	remove_variable = tiger_gender
}
